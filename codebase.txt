document.querySelectorAll('.reveal').forEach(el => observer.observe(el));

// Reviews System
const testimonialsGrid = document.getElementById('testimonialsGrid');
const emptyReviews = document.getElementById('emptyReviews');
const reviewForm = document.getElementById('reviewForm');
let selectedRating = 0;
const stars = document.querySelectorAll('#starRating .star');

function updateEmptyState() {
  const reviews = JSON.parse(localStorage.getItem('reviews') || '[]');
  emptyReviews.style.display = reviews.length === 0 ? 'block' : 'none';
}

function updateStarsDisplay(rating) {
  stars.forEach((s, index) => s.classList.toggle('active', index < rating));
}
stars.forEach(star => {
  star.addEventListener('click', () => { selectedRating = parseInt(star.getAttribute('data-rating')); updateStarsDisplay(selectedRating); });
  star.addEventListener('mouseenter', () => { const r = parseInt(star.getAttribute('data-rating')); stars.forEach((s, i) => s.style.color = i < r ? 'var(--amber)' : 'var(--border)'); });
});
document.getElementById('starRating').addEventListener('mouseleave', () => { stars.forEach(s => s.style.color = ''); updateStarsDisplay(selectedRating); });

function addReviewToDOM(review) {
  const card = document.createElement('div');
  card.className = 'testimonial-card glass';
  const initials = review.name.split(' ').map(n => n[0]).join('').toUpperCase().slice(0, 2);
  card.innerHTML = `
    <div class="testimonial-stars">${'★'.repeat(review.rating)}${'☆'.repeat(5-review.rating)}</div>
    <p class="testimonial-text">"${review.text}"</p>
    <div class="testimonial-author">
      <div class="testimonial-avatar">${initials}</div>
      <div class="testimonial-info"><h4>${review.name}</h4><p>${review.business}</p></div>
    </div>`;
  testimonialsGrid.appendChild(card);
}

const savedReviews = JSON.parse(localStorage.getItem('reviews') || '[]');
savedReviews.forEach(addReviewToDOM);
updateEmptyState();

reviewForm.addEventListener('submit', (e) => {
  e.preventDefault();
  if (selectedRating === 0) { alert('Veuillez sélectionner une note'); return; }
  const review = { name: document.getElementById('reviewName').value, business: document.getElementById('reviewBusiness').value, text: document.getElementById('reviewText').value, rating: selectedRating };
  const reviews = JSON.parse(localStorage.getItem('reviews') || '[]');
  reviews.unshift(review);
  localStorage.setItem('reviews', JSON.stringify(reviews));
  addReviewToDOM(review);
  reviewForm.reset(); selectedRating = 0; updateStarsDisplay(0); updateEmptyState();
  alert('Merci pour votre avis !');
});
